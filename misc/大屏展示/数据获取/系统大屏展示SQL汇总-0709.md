## 数据汇聚总览

主要依赖三个系统合并后的数据：

用户人员表（人员id、绑定的设备信息、状态）按照人员id进行合并

- 运维：工单sys_user表，t_repairman表，监控user表
- Z监管：人员表

组织机构表（组织id、包含人员）

- 运维：工单中的sys_dept表
- Z监管：组织结构表

应用表（关联用户信息、设备IP、MAC等信息）

- 运维：应用系统表

设备表（设备类型：包含终端、服务器、网络设备等不同类型，运维在线状态（与设备表关联）查询在线率、要查询终端有xx条，服务器有xx条，实现按比例进行分配）按照IP、MAC的方式进行合并

- 安管：设备信息表
- 自监管：设备表
- 运维：监控的hosts表（设备状态）、工单的t_resource_item

```sql
-- ===================================================================
-- 1. 终端统计

SELECT '===== 1. 终端统计 =====' AS 统计项;

SELECT 
    'AG_ASSET_DEVICE' AS 来源表,
    NVL(COUNT(*), 0) AS 终端总数,

SELECT

    NVL(COUNT(*), 0) AS 终端总数,
    NVL(SUM(CASE WHEN TERMINAL_PURPOSE = '运维' THEN 1 ELSE 0 END), 0) AS 运维用户终端数,
    NVL(SUM(CASE WHEN TERMINAL_PURPOSE IS NULL OR TERMINAL_PURPOSE = '普通' THEN 1 ELSE 0 END), 0) AS 普通用户终端数
FROM SECMON."ZJG-DEVICE"
WHERE DEVICE_TYPE = '终端';

SELECT 
    '合计' AS 来源表,
    NVL(COUNT(*), 0) AS 终端总数,
    NVL(SUM(CASE WHEN usage_type = '运维' THEN 1 ELSE 0 END), 0) AS 运维用户终端数,
    NVL(SUM(CASE WHEN usage_type = '普通' THEN 1 ELSE 0 END), 0) AS 普通用户终端数
FROM (
    SELECT CASE WHEN TERMINAL_USAGE = 1 THEN '运维' ELSE '普通' END AS usage_type
    FROM SECMON.AG_ASSET_DEVICE WHERE "TYPE" = '终端'
    UNION ALL
    SELECT CASE WHEN TERMINAL_PURPOSE = '运维' THEN '运维' ELSE '普通' END
    FROM SECMON."ZJG-DEVICE" WHERE DEVICE_TYPE = '终端'

SELECT 
    'AG_ASSET_DEVICE' AS 来源表,

FROM SECMON.AG_ASSET_DEVICE
WHERE "TYPE" = '服务器';

SELECT 
    'ZJG-DEVICE' AS 来源表,

SELECT

) n;


-- ===================================================================
-- 4. 在线资产率
-- ===================================================================
SELECT '===== 4. 在线资产率 =====' AS 统计项;

WITH 
total_devices AS (
    SELECT NVL(COUNT(*), 0) AS total_cnt FROM (
        SELECT ID FROM SECMON.AG_ASSET_DEVICE
        UNION ALL
        SELECT OBJ_DEVICE_ID FROM SECMON."ZJG-DEVICE"
        UNION ALL
        SELECT HOSTID FROM SECMON."YW-HOSTS"
    ) all_devices
),
online_devices AS (
    SELECT NVL(COUNT(*), 0) AS online_cnt
    FROM SECMON."YW-HOSTS"
    WHERE STATUS = 0
)
SELECT 
    NVL(online_cnt, 0) AS 在线设备数,
    NVL(total_cnt, 0) AS 总设备数,
    CASE 
        WHEN NVL(total_cnt, 0) = 0 THEN 0.00
        ELSE ROUND(NVL(online_cnt, 0) / NVL(total_cnt, 0) * 100, 2)
    END AS 在线资产率
FROM online_devices, total_devices;


-- ===================================================================
-- 5. 综合统计汇总（单行输出，所有值均为0而非NULL）
-- ===================================================================
SELECT '===== 5. 综合统计汇总 =====' AS 统计项;

WITH 
term AS (
    SELECT 
        NVL(COUNT(*), 0) AS term_total,
        NVL(SUM(CASE WHEN usage_type = '运维' THEN 1 ELSE 0 END), 0) AS term_ops,
        NVL(SUM(CASE WHEN usage_type = '普通' THEN 1 ELSE 0 END), 0) AS term_normal
    FROM (
        SELECT CASE WHEN TERMINAL_USAGE = 1 THEN '运维' ELSE '普通' END AS usage_type
        FROM SECMON.AG_ASSET_DEVICE WHERE "TYPE" = '终端'
        UNION ALL
        SELECT CASE WHEN TERMINAL_PURPOSE = '运维' THEN '运维' ELSE '普通' END
        FROM SECMON."ZJG-DEVICE" WHERE DEVICE_TYPE = '终端'
    ) t
),
srv AS (
    SELECT 
        NVL(COUNT(*), 0) AS srv_total,
        NVL(SUM(CASE WHEN sub_type = '物理机' THEN 1 ELSE 0 END), 0) AS srv_physical,
        NVL(SUM(CASE WHEN sub_type = '虚拟机' THEN 1 ELSE 0 END), 0) AS srv_virtual
    FROM (
        SELECT SUB_TYPE_NAME AS sub_type FROM SECMON.AG_ASSET_DEVICE WHERE "TYPE" = '服务器'
        UNION ALL
        SELECT HARDWARE_MODEL FROM SECMON."ZJG-DEVICE" WHERE DEVICE_TYPE = '服务器'
    ) s
),
net AS (
    SELECT 
        NVL(COUNT(*), 0) AS net_total,
        NVL(SUM(CASE WHEN sub_type LIKE '%路由%' THEN 1 ELSE 0 END), 0) AS net_router,
        NVL(SUM(CASE WHEN sub_type LIKE '%交换%' THEN 1 ELSE 0 END), 0) AS net_switch,
        NVL(SUM(CASE WHEN sub_type LIKE '%防火%' THEN 1 ELSE 0 END), 0) AS net_firewall
    FROM (
        SELECT SUB_TYPE_NAME AS sub_type FROM SECMON.AG_ASSET_DEVICE WHERE "TYPE" = '网络设备'
        UNION ALL
        SELECT DEVICE_TYPE FROM SECMON."ZJG-DEVICE"
        WHERE DEVICE_TYPE IN ('路由器', '交换机', '防火墙')
           OR DEVICE_TYPE LIKE '%路由%' OR DEVICE_TYPE LIKE '%交换%' OR DEVICE_TYPE LIKE '%防火%'
    ) n
),
total_dev AS (
    SELECT NVL(COUNT(*), 0) AS total_cnt FROM (
        SELECT ID FROM SECMON.AG_ASSET_DEVICE
        UNION ALL
        SELECT OBJ_DEVICE_ID FROM SECMON."ZJG-DEVICE"
        UNION ALL
        SELECT HOSTID FROM SECMON."YW-HOSTS"
    )
),
online_dev AS (
    SELECT NVL(COUNT(*), 0) AS online_cnt
    FROM SECMON."YW-HOSTS" WHERE STATUS = 0
)
SELECT 
	NVL(total_dev.total_cnt, 0)   AS 总设备数,
    NVL(term.term_total, 0) AS 终端总数,
    NVL(term.term_ops, 0)    AS 运维终端数,
    NVL(term.term_normal, 0) AS 普通用户终端数,
    NVL(srv.srv_total, 0)    AS 服务器总数,
    NVL(srv.srv_physical, 0) AS 物理机数,
    NVL(srv.srv_virtual, 0)  AS 虚拟机数,
    NVL(net.net_total, 0)    AS 网络设备总数,
    NVL(net.net_router, 0)   AS 路由器数,
    NVL(net.net_switch, 0)   AS 交换机数,
    NVL(net.net_firewall, 0) AS 防火墙数,
    NVL(online_dev.online_cnt, 0) AS 在线设备数,
    CASE 
        WHEN NVL(total_dev.total_cnt, 0) = 0 THEN 0.00
        ELSE ROUND(NVL(online_dev.online_cnt, 0) / NVL(total_dev.total_cnt, 0) * 100, 2)
    END AS 在线资产率
FROM term, srv, net, total_dev, online_dev;
```

```sql
-- 今日异常事件
SELECT
  (SELECT COUNT(*)
   FROM SECMON."YW-EVENTS"
   WHERE CLOCK >= (TRUNC(SYSDATE) - TO_DATE('1970-01-01', 'YYYY-MM-DD')) * 86400
     AND CLOCK < (TRUNC(SYSDATE + 1) - TO_DATE('1970-01-01', 'YYYY-MM-DD')) * 86400
  ) AS 运管今日告警数,

  (SELECT COUNT(*)
   FROM SECMON."AG_ALARM_INFO"
   WHERE Storage_time >= TRUNC(SYSDATE)
     AND Storage_time < TRUNC(SYSDATE + 1)
  ) AS 安管今日告警数,

  (SELECT COUNT(*)
   FROM SECMON."YW-EVENTS"
   WHERE CLOCK >= (TRUNC(SYSDATE) - TO_DATE('1970-01-01', 'YYYY-MM-DD')) * 86400
     AND CLOCK < (TRUNC(SYSDATE + 1) - TO_DATE('1970-01-01', 'YYYY-MM-DD')) * 86400
  ) + (
   SELECT COUNT(*)
   FROM SECMON."AG_ALARM_INFO"
   WHERE Storage_time >= TRUNC(SYSDATE)
     AND Storage_time < TRUNC(SYSDATE + 1)
  ) AS 今日告警总数

FROM DUAL;

   FROM SECMON."YW-EVENTS"

     AND CLOCK < (TRUNC(SYSDATE + 1) - TO_DATE('1970-01-01', 'YYYY-MM-DD')) * 86400
      

-- 异常行为本月累计
SELECT
  (SELECT COUNT(*)
   FROM SECMON."YW-EVENTS"
   WHERE CLOCK >= DATEDIFF(SS, '1970-01-01 08:00:00', TO_DATE(TO_CHAR(SYSDATE, 'YYYY-MM'), 'YYYY-MM'))
     AND CLOCK < DATEDIFF(SS, '1970-01-01 08:00:00', TRUNC(SYSDATE + 1))
  ) AS 运管本月累计,

  (SELECT COUNT(*)
   FROM SECMON."AG_ALARM_INFO"
   WHERE Storage_time >= TO_DATE(TO_CHAR(SYSDATE, 'YYYY-MM'), 'YYYY-MM')
     AND Storage_time < TRUNC(SYSDATE + 1)

     AND CLOCK < DATEDIFF(SS, '1970-01-01 08:00:00', TRUNC(SYSDATE + 1))
  ) + (
   SELECT COUNT(*)
   FROM SECMON."AG_ALARM_INFO"

    ROUND(
        ((SELECT COUNT(*) FROM SECMON."YW-EVENTS" WHERE VALUE = 0) +
         (SELECT COUNT(*) FROM SECMON."AG_ALARM_INFO" WHERE verified = 'VERIFIED_DISPOSED')) * 100.0 /
        NULLIF((SELECT COUNT(*) FROM SECMON."YW-EVENTS") +
               (SELECT COUNT(*) FROM SECMON."AG_ALARM_INFO"), 0)
    , 2) AS 告警处置率,

    -- 漏洞修复率
    ROUND(
        SUM(CASE WHEN T.CASE_STATUS = '已完成' THEN 1 ELSE 0 END) * 100.0 /
        NULLIF(COUNT(*), 0), 2
    ) AS 漏洞修复率,

    -- 合规达标率
    ROUND(
        SUM(CASE
            WHEN CAST(REGEXP_SUBSTR(T.TIME_LIMIT, '[0-9]+') AS INT)
                 >= DATEDIFF(HOUR, T.REPORT_TIME, T.END_TIME)
            THEN 1 ELSE 0
        END) * 100.0 /
        NULLIF(SUM(CASE WHEN T.CASE_STATUS = '已完成' THEN 1 ELSE 0 END), 0), 2
    ) AS 合规达标率,

    -- 综合安全指数 = 告警处置率 × 0.4 + 漏洞修复率 × 0.3 + 合规达标率 × 0.3
    ROUND(
        (
            -- 告警处置率 × 0.4
            ((SELECT COUNT(*) FROM SECMON."YW-EVENTS" WHERE VALUE = 0) +
             (SELECT COUNT(*) FROM SECMON."AG_ALARM_INFO" WHERE verified = 'VERIFIED_DISPOSED')) * 100.0 /
            NULLIF((SELECT COUNT(*) FROM SECMON."YW-EVENTS") +
                   (SELECT COUNT(*) FROM SECMON."AG_ALARM_INFO"), 0)
        ) * 0.4
        +
        (
            -- 漏洞修复率 × 0.3
            SUM(CASE WHEN T.CASE_STATUS = '已完成' THEN 1 ELSE 0 END) * 100.0 /
            NULLIF(COUNT(*), 0)
        ) * 0.3
        +
        (
            -- 合规达标率 × 0.3
            SUM(CASE
                WHEN CAST(REGEXP_SUBSTR(T.TIME_LIMIT, '[0-9]+') AS INT)
                     >= DATEDIFF(HOUR, T.REPORT_TIME, T.END_TIME)
                THEN 1 ELSE 0
            END) * 100.0 /
            NULLIF(SUM(CASE WHEN T.CASE_STATUS = '已完成' THEN 1 ELSE 0 END), 0)
        ) * 0.3
    , 2) AS 综合安全指数

FROM SECMON."YWGD-T_SERVICE_TICKETS" T;


-- 告警处置趋势总览(一周内)
WITH day_range AS (
  SELECT TRUNC(SYSDATE) - LEVEL + 1 AS dt
  FROM DUAL
  CONNECT BY LEVEL <= 7
)
SELECT
  TO_CHAR(d.dt, 'YYYY-MM-DD') AS 日期,
  (SELECT COUNT(*) FROM SECMON."YW-EVENTS"
   WHERE CLOCK >= DATEDIFF(SS, '1970-01-01 08:00:00', d.dt)
     AND CLOCK < DATEDIFF(SS, '1970-01-01 08:00:00', d.dt + 1)
  ) + (
   SELECT COUNT(*) FROM SECMON."AG_ALARM_INFO"
   WHERE Storage_time >= d.dt AND Storage_time < d.dt + 1
  ) AS 告警数,
  (SELECT COUNT(*) FROM SECMON."YW-EVENTS"
   WHERE VALUE = 0
     AND CLOCK >= DATEDIFF(SS, '1970-01-01 08:00:00', d.dt)
     AND CLOCK < DATEDIFF(SS, '1970-01-01 08:00:00', d.dt + 1)
  ) + (
   SELECT COUNT(*) FROM SECMON."AG_ALARM_INFO"
   WHERE verified = 'VERIFIED_DISPOSED'
     AND Storage_time >= d.dt AND Storage_time < d.dt + 1
  ) AS 处置数,
  NVL(
    ROUND(
      ((SELECT COUNT(*) FROM SECMON."YW-EVENTS"
        WHERE VALUE = 0
          AND CLOCK >= DATEDIFF(SS, '1970-01-01 08:00:00', d.dt)
          AND CLOCK < DATEDIFF(SS, '1970-01-01 08:00:00', d.dt + 1)
       ) + (
        SELECT COUNT(*) FROM SECMON."AG_ALARM_INFO"
        WHERE verified = 'VERIFIED_DISPOSED'
          AND Storage_time >= d.dt AND Storage_time < d.dt + 1
       )) * 100.0 /
      NULLIF(
        (SELECT COUNT(*) FROM SECMON."YW-EVENTS"
         WHERE CLOCK >= DATEDIFF(SS, '1970-01-01 08:00:00', d.dt)
           AND CLOCK < DATEDIFF(SS, '1970-01-01 08:00:00', d.dt + 1)
        ) + (
         SELECT COUNT(*) FROM SECMON."AG_ALARM_INFO"
         WHERE Storage_time >= d.dt AND Storage_time < d.dt + 1
        ), 0)
    , 2)
  , 0) AS 处置率_percent
FROM day_range d
ORDER BY d.dt;
```
